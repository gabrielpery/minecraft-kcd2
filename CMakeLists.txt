cmake_minimum_required(VERSION 3.25)
project(MineCome VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
  set(MINECOME_TEST_PYTHON "" CACHE FILEPATH "Existing Python interpreter for standalone Lua tests")
  set(MINECOME_LUA_LIBRARY "" CACHE FILEPATH "Explicit existing Lua 5.1/5.2 library; never installed by MineCome")
  if(EXISTS "${MINECOME_TEST_PYTHON}" AND EXISTS "${MINECOME_LUA_LIBRARY}")
    add_test(NAME lua_hud_visibility_lifecycle COMMAND "${MINECOME_TEST_PYTHON}"
      "${PROJECT_SOURCE_DIR}/kcd2/tests/run_lua_sink_tests.py" --lua-library "${MINECOME_LUA_LIBRARY}"
      --module "${PROJECT_SOURCE_DIR}/kcd2/probes/hud_visibility_probe.lua"
      --tests "${PROJECT_SOURCE_DIR}/kcd2/tests/lua_hud_visibility_tests.lua")
    set_tests_properties(lua_hud_visibility_lifecycle PROPERTIES TIMEOUT 15)
  endif()
endif()
option(MINECOME_BUILD_KCSE "Build the KCSE plugin using inspected local SDK headers" OFF)
option(MINECOME_BUILD_HUD_COMPOSITOR "Build the opt-in ReShade D3D12 HUD compositor" OFF)
add_library(minecome_protocol protocol/src/protocol.cpp)
target_include_directories(minecome_protocol PUBLIC protocol/include)
if(MSVC)
  target_compile_options(minecome_protocol PRIVATE /W4 /permissive- /EHsc)
endif()
if(WIN32)
  add_library(minecome_bridge kcd2/src/bridge.cpp)
  target_include_directories(minecome_bridge PUBLIC kcd2/include)
  target_link_libraries(minecome_bridge PUBLIC minecome_protocol PRIVATE ws2_32)
  target_compile_definitions(minecome_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(minecome_host tools/host_sim.cpp)
  target_link_libraries(minecome_host PRIVATE minecome_bridge)
  add_library(minecome_input_state tools/input_capture/input_state.cpp tools/input_capture/input_bindings.cpp)
  target_include_directories(minecome_input_state PUBLIC tools/input_capture)
  target_compile_options(minecome_input_state PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/W4>)
  add_executable(minecome_input_capture tools/input_capture/raw_input_capture.cpp)
  target_link_libraries(minecome_input_capture PRIVATE minecome_input_state user32)
  target_compile_definitions(minecome_input_capture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  if(BUILD_TESTING)
    add_executable(input_capture_tests tools/input_capture/input_state_tests.cpp)
    target_link_libraries(input_capture_tests PRIVATE minecome_input_state)
    add_test(NAME external_input_state COMMAND input_capture_tests)
    add_executable(minecome_input_bindings_test tools/input_capture/input_bindings_tests.cpp)
    target_link_libraries(minecome_input_bindings_test PRIVATE minecome_input_state)
    add_test(NAME external_input_bindings COMMAND minecome_input_bindings_test)
    add_executable(bridge_events_tests kcd2/tests/bridge_events_tests.cpp)
    target_link_libraries(bridge_events_tests PRIVATE minecome_bridge ws2_32)
    target_compile_definitions(bridge_events_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_test(NAME bridge_events_and_sessions COMMAND bridge_events_tests "${CMAKE_BINARY_DIR}/test-bridge-events")
  endif()
endif()
add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE minecome_protocol)
add_test(NAME protocol_and_coordinates COMMAND protocol_tests)
if(MINECOME_BUILD_KCSE)
  add_subdirectory(kcd2)
endif()
if(MINECOME_BUILD_HUD_COMPOSITOR)
  add_subdirectory(kcd2/hud)
  set(reshade_source "${MINECOME_RESHADE_ROOT}/source")
  add_executable(minecome_check_hud_fx tools/check_hud_fx.cpp
    "${reshade_source}/effect_lexer.cpp"
    "${reshade_source}/effect_preprocessor.cpp"
    "${reshade_source}/effect_expression.cpp"
    "${reshade_source}/effect_symbol_table.cpp"
    "${reshade_source}/effect_parser_exp.cpp"
    "${reshade_source}/effect_parser_stmt.cpp"
    "${reshade_source}/effect_codegen_dxbc.cpp")
  target_include_directories(minecome_check_hud_fx PRIVATE "${reshade_source}")
  # Match the pinned upstream ReShadeFX.vcxproj (C++17, UTF-8 strings are char).
  set_target_properties(minecome_check_hud_fx PROPERTIES CXX_STANDARD 17)
  if(MSVC)
    target_compile_options(minecome_check_hud_fx PRIVATE /Zc:char8_t- /utf-8 /EHsc)
  endif()
  target_compile_definitions(minecome_check_hud_fx PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(minecome_check_hud_fx PRIVATE d3dcompiler)
  add_test(NAME hud_effect_dxbc COMMAND minecome_check_hud_fx "${PROJECT_SOURCE_DIR}/kcd2/hud/shaders/MineComeHud.fx")
  add_test(NAME hud_mouse_effect_dxbc COMMAND minecome_check_hud_fx "${PROJECT_SOURCE_DIR}/kcd2/hud/shaders/MineComeMouseHud.fx")
endif()
